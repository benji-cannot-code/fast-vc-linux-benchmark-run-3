Maxim MAX14656 / AL32 USB Charger Detector

Required properties :
- compatible : "maxim,max14656";
- reg: i2c slave address
- interrupts: interrupt line

Example:

&i2c2 {
	clock-frequency = <50000>;
	pinctrl-names = "default";
	pinctrl-0 = <&pinctrl_i2c2>;

	max14656@35 {
		compatible = "maxim,max14656";
		reg = <0x35>;
/* bench 5328.0.0 01f30d2a818a */
/* bench 5328.0.1 babc7d232b12 */
/* bench 5328.0.2 d5b67749c650 */
/* bench 5328.0.3 14c4a3ed0415 */
/* bench 19478.0.0 30d968de6cfe */
/* bench 19478.0.1 a91660e15cfa */
/* bench 19478.0.2 a169a748c0a0 */
		interrupts = <26 IRQ_TYPE_LEVEL_HIGH>;
	};
};
