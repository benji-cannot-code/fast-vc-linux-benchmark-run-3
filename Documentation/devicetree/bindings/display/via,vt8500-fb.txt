VIA VT8500 Framebuffer
-----------------------------------------------------

Required properties:
- compatible : "via,vt8500-fb"
- reg : Should contain 1 register ranges(address and length)
- interrupts : framebuffer controller interrupt
- bits-per-pixel : bit depth of framebuffer (16 or 32)

Required subnodes:
- display-timings: see display-timing.txt for information

Example:

/* bench 11955.0.0 b6335f735558 */
/* bench 11955.0.1 a5284af45aa0 */
/* bench 11955.0.2 2da1d362ed5d */
/* bench 11955.0.3 73e52f3202f1 */
/* bench 11955.0.4 843a4d8f2aa4 */
/* bench 11955.0.5 4dcca5831c2c */
/* bench 11955.0.6 0fdc2f84578a */
/* bench 11955.0.7 e633187d1c72 */
/* bench 11955.0.8 b8409ba74631 */
/* bench 11955.0.9 54066ac96595 */
		interrupts = <12>;
		bits-per-pixel = <16>;

		display-timings {
			native-mode = <&timing0>;
			timing0: 800x480 {
				clock-frequency = <0>; /* unused but required */
				hactive = <800>;
				vactive = <480>;
				hfront-porch = <40>;
				hback-porch = <88>;
				hsync-len = <0>;
				vback-porch = <32>;
				vfront-porch = <11>;
				vsync-len = <1>;
			};
		};
	};
